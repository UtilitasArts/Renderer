void load_obj_file_data(const char *filename) {
    namespace fs = std::filesystem;

    fs::path Folder = SDL_GetBasePath();
    while(Folder.parent_path().filename().string() != "Renderer"){
        Folder = Folder.parent_path();
    }
    
    fs::path FilePath = Folder.parent_path().string() + "/assets/" + std::string(filename) + ".obj";  

    if(!FilePath.has_filename()) {
        std::cout << "No such file In Asset Folder: " << std::endl;       
        return;
    }
    else{
         std::cout << "File Name: " << FilePath.filename().string() << std::endl;
    }

    std::ifstream File;    
    File.open(FilePath);

    std::string Content;

    std::regex type_pattern{"v|f"};   
    std::regex v_pattern{"v (.*) (.*) (.*)"}; 
    std::regex f_pattern{"f (\\d*)/(\\d*)/(\\d*) (\\d*)/(\\d*)/(\\d*) (\\d*)/(\\d*)/(\\d*)"}; 

    std::smatch type_match; 
    std::smatch v_match; 
    std::smatch f_match; 

    while(std::getline(File,Content)) {

        // std::cout << Content << std::endl;
        std::regex_search(Content, type_match,type_pattern);

        if (type_match[0]=="v"){
           
            std::regex_search(Content, v_match ,v_pattern);

            if (v_match.size() == 4){
                // std::cout << Match2[1].str() << Match2[2].str() << Match2[3].str() << std::endl ;
                float x = std::stof(v_match[1].str());
                float y = std::stof(v_match[2].str());
                float z = std::stof(v_match[3].str());  

                vec3_t vertex{.x = x, .y = y,.z = z};
                mesh.vertices.push_back(vertex);
            }
        }
        if (type_match[0] == "f"){
            std::regex_search(Content, f_match ,f_pattern);

            if (f_match.size() == 10){
                // std::cout << Match2[1].str() << Match2[2].str() << Match2[3].str() << std::endl ;
                int v1 = std::stoi(f_match[1].str());
                int v2 = std::stoi(f_match[4].str());
                int v3 = std::stoi(f_match[7].str());  

                face_t face{.a = v1, .b = v2, .c = v3};
                mesh.faces.push_back(face);          
            }
        }         
    }
    

    File.close();
